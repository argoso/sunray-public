};

  //{PE6, "Startup LOW"}
};

// -----------------------------
// Helper functions
// -----------------------------
String getPinName(int pin) {
  for (int i=0;i<totalPinCount;i++)
    if (pinMap[i].pin==pin) return pinMap[i].name;
  return "?";
}

int getPinByName(String name) {
  name.trim(); name.toUpperCase();
  for (int i=0;i<totalPinCount;i++)
    if (name == pinMap[i].name) return pinMap[i].pin;
  return -1;
}

bool isExcluded(int pin) {
  for (int i=0;i<excludedCount;i++)
    if (excludedPins[i].pin==pin) return true;
  return false;
}

int getPinIndex(int pin) {
  for (int i=0;i<totalPinCount;i++)
    if (pinMap[i].pin == pin)
      return i;
  return -1;
}

// -----------------------------
// Scan state
// -----------------------------
enum ScanState { SCAN, PAUSED, STOPPED };
ScanState scanState = SCAN;

int currentIndex = 0;
bool stopFlag = false;

unsigned long testLowTime = 5000;       // 3 sek LOW
unsigned long startupHighDelay = 5000;  // 5 sek kõik HIGH
bool startupDone = false;

// -----------------------------
// Pin test (LOW → HIGH)
// -----------------------------
void testPin(int pin) {
  Console.println();
  Console.print("Testin pinni: ");
  Console.println(getPinName(pin));

  pinMode(pin, OUTPUT);

  Console.println("  LOW (3 sek)...");
  digitalWrite(pin, LOW);
  delay(testLowTime);

  Console.println("  HIGH tagasi...");
  digitalWrite(pin, HIGH);
  delay(200);

  Console.println("  >> Test loppenud!");
  Console.println();
}

// -----------------------------
// All non excluded pins HIGH state at begin
// -----------------------------
void setAllPinsHighExceptExcluded() {
  Console.println("Seadistan koik mitte-excluded pinid HIGH...");

  for (int i = 0; i < totalPinCount; i++) {
    int pin = pinMap[i].pin;

    if (!isExcluded(pin)) {
      pinMode(pin, OUTPUT);
      digitalWrite(pin, HIGH);

      Console.print("  HIGH -> ");
      Console.println(getPinName(pin));
    }
  }

  Console.print("Ootan ");
  Console.print(startupHighDelay / 1000);
  Console.println(" sekundit...");
  delay(startupHighDelay);

  Console.println("Algus HIGH tehtud!");
}

// -----------------------------
// Scan function
// -----------------------------
void scanPins() {
  if (currentIndex >= totalPinCount) {
    Console.println(">>> Kõik pinid testitud!");
    scanState = STOPPED;
    return;
  }

  int pin = pinMap[currentIndex].pin;

  if (isExcluded(pin)) {
    currentIndex++;
    return;
  }

  Console.print("Testin PIN: ");
  Console.println(getPinName(pin));

  testPin(pin);
  currentIndex++;
}

// -----------------------------
// Serial commands
// -----------------------------
void processSerialCommand(char cmd) {
  switch (cmd) {
    case 'p': case 'P':
      scanState = PAUSED;
      Console.println("PAUSE");
      break;

    case 'n': case 'N':
      if (scanState==PAUSED) {
        scanState = SCAN;
        Console.println("NEXT");
      }
      break;

    case 'r': case 'R':
      currentIndex = 0;
      stopFlag = false;
      scanState = SCAN;
      Console.println("RESTART");
      break;

    case 'm': case 'M': {
      Console.println("Sisesta pin nimi:");
      while (!Console.available()) delay(10);
      String p = Console.readStringUntil('\n');
      p.trim();
      int pin = getPinByName(p);
      if (pin != -1 && !isExcluded(pin))
        testPin(pin);
      else
        Console.println("Vigane vagy excluded pin");
      break;
    }

    default:
      Console.println("Käsud: p=PAUSE, n=NEXT, r=RESTART, m=MANUAL");
      break;
  }
}

// -----------------------------
// setup
// -----------------------------
void setup() {
  Console.begin(115200);
  delay(1500);

  Console.println("MC33035 PIN MAPPER — auto HIGH start");

  // Initial HIGH/LOW (sinu manuaalsed sätted)
  for (int i=0;i<initialHighCount;i++) {
    pinMode(initialHighPins[i].pin, OUTPUT);
    digitalWrite(initialHighPins[i].pin, HIGH);
  }
  for (int i=0;i<initialLowCount;i++) {
    pinMode(initialLowPins[i].pin, OUTPUT);
    digitalWrite(initialLowPins[i].pin, LOW);
  }

  // Kõik teised pinid HIGH (ainult 1x)
  if (!startupDone) {
    setAllPinsHighExceptExcluded();
    startupDone = true;
  }

  scanState = SCAN;
}

// -----------------------------
// loop
// -----------------------------
void loop() {
  if (Console.available())
    processSerialCommand(Console.read());

  if (scanState == SCAN && !stopFlag)
    scanPins();
  else
    delay(50);
}
